**代码1：手写 Scaled Dot-Product Attention**
```python
import torch
import torch.nn.functional as F

def scaled_dot_product_attention(Q, K, V, mask=None):
    d_k = Q.size(-1)
    scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(d_k)
    if mask is not None:
        scores = scores.masked_fill(mask == 0, float('-inf'))
    attn_weights = F.softmax(scores, dim=-1)
    output = torch.matmul(attn_weights, V)
    return output, attn_weights
```

**代码2：LoRA 层前向传播**
```python
class LoRALinear(nn.Module):
    def __init__(self, in_features, out_features, r=8, alpha=16, dropout=0.0):
        super().__init__()
        self.linear = nn.Linear(in_features, out_features, bias=False)
        self.lora_A = nn.Parameter(torch.randn(r, in_features) * 0.02)
        self.lora_B = nn.Parameter(torch.zeros(out_features, r))
        self.scaling = alpha / r
        self.dropout = nn.Dropout(dropout)
    
    def forward(self, x):
        base = self.linear(x)
        lora_out = (self.dropout(x) @ self.lora_A.T @ self.lora_B.T) * self.scaling
        return base + lora_out
```

### 必写代码（3段）

**代码1：DPO loss（从零实现）**
```python
def dpo_loss(policy_chosen_logps, policy_rejected_logps,
             ref_chosen_logps, ref_rejected_logps, beta=0.1):
    """
    policy_chosen_logps: log π_θ(y_w|x), 标量或 per-token 的 sum
    """
    policy_log_ratio = policy_chosen_logps - policy_rejected_logps
    ref_log_ratio = ref_chosen_logps - ref_rejected_logps
    
    logits = policy_log_ratio - ref_log_ratio
    losses = -F.logsigmoid(beta * logits)
    
    chosen_rewards = beta * (policy_chosen_logps - ref_chosen_logps).detach()
    rejected_rewards = beta * (policy_rejected_logps - ref_rejected_logps).detach()
    reward_acc = (chosen_rewards > rejected_rewards).float().mean()
    
    return losses.mean(), reward_acc
```

**代码1：简单的向量检索流程**
```python
import faiss
from sentence_transformers import SentenceTransformer

class SimpleRetriever:
    def __init__(self, model_name="BAAI/bge-large-zh-v1.5"):
        self.encoder = SentenceTransformer(model_name)
        self.index = None
        self.chunks = []
    
    def build_index(self, documents, chunk_size=256, overlap=50):
        from langchain.text_splitter import RecursiveCharacterTextSplitter
        splitter = RecursiveCharacterTextSplitter(
            chunk_size=chunk_size, chunk_overlap=overlap)
        self.chunks = splitter.split_documents(documents)
        embeddings = self.encoder.encode(
            [c.page_content for c in self.chunks], normalize_embeddings=True)
        self.index = faiss.IndexFlatIP(embeddings.shape[1])
        self.index.add(embeddings.astype('float32'))
    
    def retrieve(self, query, top_k=5):
        q_emb = self.encoder.encode([query], normalize_embeddings=True)
        scores, indices = self.index.search(q_emb.astype('float32'), top_k)
        return [(self.chunks[i].page_content, scores[0][j]) 
                for j, i in enumerate(indices[0])]
```

**代码2：Safety-RAG 验证逻辑伪代码**
```python
class SafetyRAGVerifier:
    def __init__(self, llm, retriever):
        self.llm = llm
        self.retriever = retriever
    
    def verify(self, query, model_response):
        # Step 1: 检索相关知识
        retrieved = self.retriever.retrieve(query, top_k=5)
        
        # Step 2: 让 LLM 比对 response 与 retrieved knowledge
        verification_prompt = f"""
        请比对以下模型回答与医学知识库内容是否一致：
        模型回答：{model_response}
        知识库参考：{retrieved}
        请指出：
        1. 是否存在事实错误
        2. 是否存在幻觉（编造不存在的信息）
        3. 安全等级（safe / needs_revision / unsafe）
        """
        verdict = self.llm.generate(verification_prompt)
        
        # Step 3: 根据验证结果决定是否修正
        if "unsafe" in verdict or "needs_revision" in verdict:
            corrected = self.llm.generate(
                f"基于以下知识重新回答：\n知识：{retrieved}\n问题：{query}")
            return corrected, verdict, retrieved
        return model_response, verdict, retrieved
```

```python
def llm_judge_eval(judge_model, tokenizer, prompt, response_a, response_b):
    judge_template = """请对比以下两个回答，从准确性、安全性、完整性三个维度评判。
    
问题：{question}

回答A：{answer_a}

回答B：{answer_b}

1. 每个维度的对比（A更好/B更好/持平）
2. 总体判断（A更好/B更好/持平）

"""
    input_text = judge_template.format(
        question=prompt, answer_a=response_a, answer_b=response_b)
    inputs = tokenizer(input_text, return_tensors="pt")
    outputs = judge_model.generate(**inputs, max_new_tokens=256)
```

```python
def sft_collate_fn(batch, tokenizer, max_length=2048, pad_token_id=0):
    prompts = [item["prompt"] for item in batch]
    responses = [item["response"] for item in batch]
    
    full_texts = [p + r for p, r in zip(prompts, responses)]
    tokenized = tokenizer(
        full_texts, max_length=max_length, truncation=True,
        padding=True, return_tensors="pt"
    )
    
    # 构造 labels：prompt 部分设为 -100
    labels = tokenized["input_ids"].clone()
    prompt_lengths = [len(tokenizer.encode(p, add_special_tokens=False)) 
                      for p in prompts]
    for i, plen in enumerate(prompt_lengths):
        labels[i, :plen] = -100
    
    return {
        "input_ids": tokenized["input_ids"],
        "attention_mask": tokenized["attention_mask"],
        "labels": labels,
    }
```

**代码3：手写完整的训练循环（PyTorch 原生）**
```python
def train_epoch(model, dataloader, optimizer, scheduler, device):
    model.train()
    total_loss = 0
    progress_bar = tqdm(dataloader)
    
    for step, batch in enumerate(progress_bar):
        input_ids = batch["input_ids"].to(device)
        attention_mask = batch["attention_mask"].to(device)
        labels = batch["labels"].to(device)
        
        outputs = model(
            input_ids=input_ids,
            attention_mask=attention_mask,
            labels=labels
        )
        loss = outputs.loss
        
        # 梯度累积（假设 accumulation_steps=4）
        loss = loss / 4
        loss.backward()
        
        if (step + 1) % 4 == 0:
            torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
            optimizer.step()
            scheduler.step()
            optimizer.zero_grad()
        
        total_loss += loss.item() * 4
        progress_bar.set_postfix({"loss": loss.item() * 4})
    
    return total_loss / len(dataloader)
```
